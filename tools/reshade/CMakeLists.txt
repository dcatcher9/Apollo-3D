# MSVC returns even this eight-byte C++ handle via a hidden member-function result
# parameter. MinGW uses a different aggregate return convention. Follow the
# explicit aggregate-return wrapper used by MinGW's native D3D12 COM headers.
# Preserve virtual slots and adapt aggregate-return getters in a generated copy.
set(SUNSHINE_RESHADE_INCLUDE_DIR "${RESHADE_SDK_DIR}/include")
if(MINGW)
  file(READ "${RESHADE_SDK_DIR}/include/reshade_api_device.hpp" reshade_device_header)
  string(REPLACE "\r\n" "\n" reshade_device_header "${reshade_device_header}")
  string(SHA256 reshade_device_hash "${reshade_device_header}")
  if(NOT reshade_device_hash STREQUAL "a716c44b6fbb064dbbb0826687728f21aa71ed45e13564e4dc1cdf36da5bdfa0")
    message(FATAL_ERROR "The MinGW ABI adapter requires the exact pinned ReShade 6.8.0 device header")
  endif()
  set(reshade_original_getter "virtual resource get_resource_from_view(resource_view view) const = 0;")
  string(REGEX MATCHALL "virtual resource get_resource_from_view\\(resource_view view\\) const = 0" reshade_getter_matches "${reshade_device_header}")
  list(LENGTH reshade_getter_matches reshade_getter_count)
  if(NOT reshade_getter_count EQUAL 1)
    message(FATAL_ERROR "Expected exactly one ReShade resource-handle getter")
  endif()
  set(SUNSHINE_RESHADE_INCLUDE_DIR "${CMAKE_CURRENT_BINARY_DIR}/sdk-mingw")
  file(MAKE_DIRECTORY "${SUNSHINE_RESHADE_INCLUDE_DIR}")
  file(GLOB reshade_sdk_headers "${RESHADE_SDK_DIR}/include/*.hpp")
  foreach(reshade_sdk_header IN LISTS reshade_sdk_headers)
    get_filename_component(reshade_sdk_header_name "${reshade_sdk_header}" NAME)
    if(NOT reshade_sdk_header_name STREQUAL "reshade_api_device.hpp")
      configure_file("${reshade_sdk_header}" "${SUNSHINE_RESHADE_INCLUDE_DIR}/${reshade_sdk_header_name}" COPYONLY)
    endif()
  endforeach()
  string(REPLACE "${reshade_original_getter}"
    "virtual resource *get_resource_from_view_abi(resource *result, resource_view view) const = 0;\n\t\tresource get_resource_from_view(resource_view view) const { resource result {}; get_resource_from_view_abi(&result, view); return result; }"
    reshade_device_header "${reshade_device_header}")
  set(reshade_original_desc_getter "virtual resource_desc get_resource_desc(resource resource) const = 0;")
  string(FIND "${reshade_device_header}" "${reshade_original_desc_getter}" reshade_desc_getter_position)
  if(reshade_desc_getter_position EQUAL -1)
    message(FATAL_ERROR "Expected the pinned ReShade resource-description getter")
  endif()
  string(REPLACE "${reshade_original_desc_getter}"
    "virtual resource_desc *get_resource_desc_abi(resource_desc *result, resource resource) const = 0;\n\t\tresource_desc get_resource_desc(resource value) const { resource_desc result {}; get_resource_desc_abi(&result, value); return result; }"
    reshade_device_header "${reshade_device_header}")
  set(reshade_original_view_desc_getter "virtual resource_view_desc get_resource_view_desc(resource_view view) const = 0;")
  string(FIND "${reshade_device_header}" "${reshade_original_view_desc_getter}" reshade_view_desc_position)
  if(reshade_view_desc_position EQUAL -1)
    message(FATAL_ERROR "Expected the pinned ReShade resource-view-description getter")
  endif()
  string(REPLACE "${reshade_original_view_desc_getter}"
    "virtual resource_view_desc *get_resource_view_desc_abi(resource_view_desc *result, resource_view view) const = 0;\n\t\tresource_view_desc get_resource_view_desc(resource_view value) const { resource_view_desc result {}; get_resource_view_desc_abi(&result, value); return result; }"
    reshade_device_header "${reshade_device_header}")
  set(reshade_adapted_device "${SUNSHINE_RESHADE_INCLUDE_DIR}/reshade_api_device.hpp")
  set(reshade_previous_device "")
  if(EXISTS "${reshade_adapted_device}")
    file(READ "${reshade_adapted_device}" reshade_previous_device)
  endif()
  if(NOT reshade_previous_device STREQUAL reshade_device_header)
    file(WRITE "${reshade_adapted_device}" "${reshade_device_header}")
  endif()
endif()
